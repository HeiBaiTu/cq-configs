## 对话配置示例

### 1. 基础对话配置 (talk/M0001.txt)

```lua
-- 怪物发言配置
talk = 
{  
    minInterval = 60, -- 下限间隔时间(只有空闲发言有效)
    maxInterval = 180, -- 上限间隔时间(只有空闲发言有效)
    
    info = 
    {
        { 
            type=6, rate=100,  -- 创建时发言
            param=0,
            bctype=1,  -- 附近广播
            content=
            {
                OldLang.Paopao.yd00001,  -- 引用语言文件
            },
        },
        { 
            type=0, rate=100,  -- 空闲状态发言
            param=0,
            bctype=1,
            content=
            {
                OldLang.Paopao.yd00001,
                OldLang.Paopao.yd00002,
                OldLang.Paopao.yd00003,
                OldLang.Paopao.yd00004,
            },
        },
        {  
            type=1, rate=100,  -- 发现目标发言
            param=0,
            bctype=1,
            content=
            {
                OldLang.Paopao.yd00005,
                OldLang.Paopao.yd00006,
                OldLang.Paopao.yd00007,
                OldLang.Paopao.yd00008,
            },
        },
        {  
            type=5, rate=50,  -- 血量低于20%发言
            param=20,
            bctype=2,  -- 场景广播
            content=
            {
                OldLang.Paopao.yd00009,
                OldLang.Paopao.yd00010,
            },
        },
    },
},
```

```lua
-- Monster.config 模板
[ID] = {
    entityid = ID,
    entityType = 1,  -- 普通怪物
    monsterType = 1, -- 普通类型
    name = "怪物名称",
    level = 等级,
    circle = 0,
    aiConfigId = AI配置ID,
    propid = 属性配置ID,
    flagid = 标志配置ID,
    exp = 经验值,
    attackInterval = 攻击间隔,
    skills = {},  -- 技能列表
    modelid = 模型ID,
    attackMusicRate = 100,
    dieMusicRate = 100,
    targetMusicTate = 100,
    damageMusicRate = 100,

-- Props.config 模板
[ID] = {
    id = ID,
```
