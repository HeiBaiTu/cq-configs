# 图片资源查找实用脚本

## 概述

本文档提供查找游戏图片资源的实用脚本和命令，帮助快速定位和管理游戏中的图片资源。

## 基础查找脚本

### 1. 查找图标资源

```bash
#!/bin/bash
# 图标资源查找脚本

# 函数：查找图标
find_icon() {
    local icon_id=$1
    echo "=== 查找图标ID: $icon_id ==="
    
    # 在物品配置中查找
    echo "物品配置 (StdItems.config):"
    if grep -q "icon.*$icon_id" data/config/item/StdItems.config; then
        grep -B 5 -A 5 "icon.*$icon_id" data/config/item/StdItems.config
    else
        echo "未找到"
    fi
    
    # 在前端配置中查找
    echo -e "\n前端配置 (StdItems.json):"
    if grep -q "icon.*$icon_id" split_configs/StdItems.json; then
        grep -B 5 -A 5 "icon.*$icon_id" split_configs/StdItems.json
    else
        echo "未找到"
    fi
}

# 使用示例
find_icon 13400
```

### 3. 查找资源文件

```bash
#!/bin/bash
# 资源文件查找脚本

# 函数：查找资源文件
find_resource_file() {
    local resource_id=$1
    local resource_type=$2  # icon, model, texture
    
    echo "=== 查找资源文件: $resource_id ($resource_type) ==="
    
    case $resource_type in
        "icon")
            # 查找图标文件
            find . -name "icon_$resource_id.*" -type f 2>/dev/null
            find . -name "*$resource_id*" -name "*.png" -type f 2>/dev/null
            find . -name "*$resource_id*" -name "*.jpg" -type f 2>/dev/null
            ;;
        "model")
            # 查找模型文件
            find . -name "monster_$resource_id.*" -type f 2>/dev/null
            find . -name "*$resource_id*" -name "*.3ds" -type f 2>/dev/null
            find . -name "*$resource_id*" -name "*.obj" -type f 2>/dev/null
            ;;
        "texture")
            # 查找贴图文件
            find . -name "texture_$resource_id.*" -type f 2>/dev/null
            find . -name "*$resource_id*" -name "*.png" -type f 2>/dev/null
            find . -name "*$resource_id*" -name "*.jpg" -type f 2>/dev/null
            ;;
    esac
}

# 使用示例
find_resource_file 13400 icon
find_resource_file 32000 model
find_resource_file 32000 texture
```

```bash
#!/bin/bash

batch_find_icons() {
    local start_id=$1
    local end_id=$2
    
    echo "=== 批量查找图标ID: $start_id - $end_id ==="
    
    for ((i=start_id; i<=end_id; i++)); do
        if grep -q "icon.*$i" data/config/item/StdItems.config; then
            echo "找到图标ID: $i"
            grep "icon.*$i" data/config/item/StdItems.config
        fi
    done
}
```

## 资源管理脚本

### 1. 资源备份脚本

```bash
#!/bin/bash
# 资源备份脚本

# 函数：备份图标资源
backup_icons() {
    local backup_dir="backup/icons/$(date +%Y%m%d_%H%M%S)"
    mkdir -p "$backup_dir"
    
    echo "备份图标资源到: $backup_dir"
    
    # 备份配置文件
    cp data/config/item/StdItems.config "$backup_dir/"
    cp split_configs/StdItems.json "$backup_dir/"
    
    # 备份图标文件
    find . -name "icon_*.png" -type f -exec cp {} "$backup_dir/" \;
    find . -name "icon_*.jpg" -type f -exec cp {} "$backup_dir/" \;
    
    echo "备份完成"
}

# 函数：备份模型资源
backup_models() {
    local backup_dir="backup/models/$(date +%Y%m%d_%H%M%S)"
    mkdir -p "$backup_dir"
    
    echo "备份模型资源到: $backup_dir"
    
    # 备份配置文件
    cp data/config/monster/Monster.config "$backup_dir/"
    cp split_configs/Monster.json "$backup_dir/"
    
    # 备份模型文件
    find . -name "monster_*.3ds" -type f -exec cp {} "$backup_dir/" \;
    find . -name "monster_*.obj" -type f -exec cp {} "$backup_dir/" \;
    
    echo "备份完成"
}

# 使用示例
backup_icons
backup_models
```

### 2. 资源清理脚本

```bash
#!/bin/bash
# 资源清理脚本

# 函数：清理未使用的图标
cleanup_unused_icons() {
    echo "清理未使用的图标..."
    
    # 获取所有配置中使用的图标ID
    grep "icon" data/config/item/StdItems.config | \
    cut -d'=' -f2 | cut -d',' -f1 | sort | uniq > used_icons.txt
    
    # 获取所有图标文件
    find . -name "icon_*.png" -type f | \
    sed 's/.*icon_\([0-9]*\)\.png/\1/' | sort > all_icons.txt
    
    # 找出未使用的图标
    comm -23 all_icons.txt used_icons.txt > unused_icons.txt
    
    # 删除未使用的图标
    while read icon_id; do
        find . -name "icon_$icon_id.*" -type f -delete
        echo "删除未使用的图标: $icon_id"
    done < unused_icons.txt
    
    # 清理临时文件
    rm -f used_icons.txt all_icons.txt unused_icons.txt
    
    echo "清理完成"
}

# 函数：清理未使用的模型
cleanup_unused_models() {
    echo "清理未使用的模型..."
    
    # 获取所有配置中使用的模型ID
    grep "modelid" data/config/monster/Monster.config | \
    cut -d'=' -f2 | cut -d',' -f1 | sort | uniq > used_models.txt
    
    # 获取所有模型文件
    find . -name "monster_*.3ds" -type f | \
    sed 's/.*monster_\([0-9]*\)\.3ds/\1/' | sort > all_models.txt
    
    # 找出未使用的模型
    comm -23 all_models.txt used_models.txt > unused_models.txt
    
    # 删除未使用的模型
    while read model_id; do
        find . -name "monster_$model_id.*" -type f -delete
        echo "删除未使用的模型: $model_id"
    done < unused_models.txt
    
    # 清理临时文件
    rm -f used_models.txt all_models.txt unused_models.txt
    
    echo "清理完成"
}

# 使用示例
cleanup_unused_icons
cleanup_unused_models
```
