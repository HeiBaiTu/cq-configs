function TestAsyncWork(paramPack,ret,str)
	local expect = paramPack.expect
	if ret == expect then
		print("AsyncWorker Test OK!")
		print("AsyncWorker ret="..ret.." str="..str)
	end
end

function TestCDKEY(paramPack,ret,numb)

	print("CDKey Test OK!")
	print("CDKey str="..ret)
	print("CDKey numb="..numb)

end
